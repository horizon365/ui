Nuxt UI v4は大きなマイルストーンとなりました。** Nuxt UIとNuxt UI Proは、完全にオープンソースで無料のライブラリ**に統合されました。125以上のプロダクション対応コンポーネントにアクセスでき、すべて`@nuxt/ui`パッケージで利用できます。

::note
Nuxt UI v4は、依存関係のため** Nuxt 4.1以降の**が必要です。Nuxt UI v4に移行する前に必ずアップグレードしてください。
::

このガイドでは、アプリケーションをv 3からv 4に移行するためのステップバイステップの手順を説明します。

## プロジェクトの移行

##  v3からの変更点

Nuxt UI v4にアップグレードした後、以下の重要な変更点に注意してください。

###  Renamed ButtonGroup

`ButtonGroup`コンポーネントは、[`FieldGroup`](/docs/components/field-group)に改名されました。

```diff
<template>
- <UButtonGroup>
+ <UFieldGroup>
    <UButton label="Button" />
    <UButton icon="i-lucide-chevron-down" />
+ </UFieldGroup>
- </UButtonGroup>
</template>
```

###  Renamed PageMarquee

`PageMarquee`コンポーネントは、[`Marquee`](/docs/components/marquee)に改名されました。

```diff
<template>
- <UPageMarquee :items="items" />
+ <UMarquee :items="items" />
</template>
```

### 削除されたPageAccordion

`PageAccordion`コンポーネントは削除され、[`Accordion`](/docs/components/accordion)に置き換えられました。

```diff
<template>
- <UPageAccordion
+ <UAccordion
    :items="items"
+   :unmount-on-hide="false"
+   :ui="{ trigger: 'text-base', body: 'text-base text-muted' }"
  />
</template>
```

::note
`PageAccordion`コンポーネントは、`unmount-on-hide`を`false`に設定し、`ui` propをカスタマイズするラッパーです。
::

### 名前変更モデル修飾子

[`Input`](/docs/components/input)[`InputNumber`](/docs/components/input-number PH27 [`Textarea`](/docs/components/textarea)で使用される`modelModifiers``modelModifiers``modelModifiers` PH2222212121

1. `nullify`修飾子が`nullable`に改名されました（空/空白値を`null`に変換します）。
2. 新しい`optional`修飾子が追加されました（空/空白の値を`undefined`に変換します）。

```diff
- <UInput v-model.nullify="value" />
+ <UInput v-model.nullable="value" />
```

```diff
- <UTextarea v-model="value" :model-modifiers="{ nullify: true }" />
+ <UTextarea v-model="value" :model-modifiers="{ nullable: true }" />
```

空の値を`null`として使用したい場合は`nullable`を使用し、存在しない値を`undefined`として使用したい場合は`optional`を使用します。

### フォームコンポーネントの変更

`Form`コンポーネントはv4で改善され、状態管理とネストされたフォーム処理が改善されました。注意すべき主な変更点は次のとおりです。

1.  Schema **変換は**`@submit` data **にのみ適用され、フォームの状態を変更しなくなります。これにより予測可能性が向上し、予期しない状態の変異を防ぎます。
2. **ネストされたフォームは、`nested` propを使用して明示的に**を有効にする必要があります。これにより、コンポーネントの動作がより明確になり、偶発的なネストされたフォームの作成を防ぎます。
3. **ネストされたフォームは、`name`** prop `UFormField`に似ていますを提供する必要があり、自動的に親フォームから状態を継承します。

```diff
<template>
  <UForm :state="state" :schema="schema" @submit="onSubmit">
    <UFormField label="Customer" name="customer">
      <UInput v-model="state.customer" placeholder="Wonka Industries" />
    </UFormField>

    <div v-for="(item, index) in state.items" :key="index">
      <UForm
-       :state="item"
+       :name="`items.${index}`"
        :schema="itemSchema"
+       nested
      >
        <UFormField :label="!index ? 'Description' : undefined" name="description">
          <UInput v-model="item.description" />
        </UFormField>
        <UFormField :label="!index ? 'Price' : undefined" name="price">
          <UInput v-model="item.price" type="number" />
        </UFormField>
      </UForm>
    </div>
  </UForm>
</template>
```

### 削除された非推奨ユーティリティ

以前Nuxt UI Proで利用可能だった** Nuxtコンテンツユーティリティ**の一部は、v4で**削除されました。

- `findPageBreadcrumb`
- `findPageHeadline`

これらはNuxt Contentによって完全に提供されます。インポートと使用方法を適宜更新してください。

```diff
- import { findPageHeadline } from '@nuxt/ui-pro/utils/content'
+ import { findPageHeadline } from '@nuxt/content/utils'

- import { findPageBreadcrumb } from '@nuxt/ui-pro/utils/content'
+ import { findPageBreadcrumb } from '@nuxt/content/utils'
```

###  AI SDK移行（オプション）

このセクションは、AI SDKとチャットコンポーネント（`ChatMessage`、`ChatMessages`、`ChatPrompt`、`ChatPromptSubmit`、`ChatPalette`）を使用している場合にのみ適用されます。AI機能を使用していない場合は、このセクションをスキップしてください。

1. `package.json`の`@ai-sdk/vue`と`ai`の依存関係を更新し、[ Comark ](https://comark.dev)を追加しますストリーミングMarkdownとしてAIレスポンスをレンダリングするために`@nuxtjs/mdc`を置き換えます

::framework-only
#nuxt
:::div
```diff
{
  "dependencies": {
-   "@ai-sdk/vue": "^1.2.x",
+   "@ai-sdk/vue": "^4.0.x",
-   "ai": "^4.3.x",
+   "ai": "^7.0.x",
+   "@comark/nuxt": "latest"
  }
}
```

モジュールに`@comark/nuxt`を追加します。

```diff [nuxt.config.ts]

    '@nuxt/ui',
+   '@comark/nuxt'
  ]
})
```
:::

#vue
```diff

  "dependencies": {
-   "@ai-sdk/vue": "^1.2.x",
+   "@ai-sdk/vue": "^4.0.x",
-   "ai": "^4.3.x",
+   "ai": "^7.0.x",
+   "@comark/vue": "latest"
```

2. `useChat`構成可能なAPIが変更されました。`input`と`handleSubmit`は削除され、`sendMessage`に置き換えられました。`messages`は直接設定するrefになりました。

```diff
<script setup lang="ts">
  import { useChat } from '@ai-sdk/vue'

- const { messages, input, handleSubmit, status, error, reload, setMessages } = useChat()
+ const { messages, status, error, sendMessage, regenerate, stop } = useChat()
+ const input = ref('')
+
+ function handleSubmit() {
+   sendMessage({ text: input.value })
+   input.value = ''
+ }
</script>
```

3. メッセージが`content`の代わりに`parts`を使用するようになりました。

```diff
// When manually creating messages
- setMessages([{
+ messages.value = [{
  id: '1',
  role: 'user',
- content: 'Hello world'
+ parts: [{ type: 'text', text: 'Hello world' }]
- }])
+ }]

// In templates
<template>
- <UChatMessage :content="message.content" />
+ <UChatMessage :parts="message.parts" />
</template>
```

4. いくつかのメソッドが名前変更されました

```diff
// Regenerate the last message
- reload()
+ regenerate()
```

5. `<MDC>`を`<Markdown>`に置き換えてください。`Markdown`コンポーネントはMarkdownのストリーミング用に作られており、AIから到着したトークンを段階的にレンダリングします。AI SDKヘルパーと`@nuxt/ui/utils/ai`の`isPartStreaming`ユーティリティを使用してください。

::note
`shiki`プラグインを使用する場合は、ダークモードをサポートするためにスタイルシートに次のCSSを追加します。

```css [main.css]
html.dark .shiki span {
  color: var(--shiki-dark) !important;
  background-color: var(--shiki-dark-bg) !important;
  font-style: var(--shiki-dark-font-style) !important;
  font-weight: var(--shiki-dark-font-weight) !important;
  text-decoration: var(--shiki-dark-text-decoration) !important;
}
```
::

::framework-only
#nuxt
```vue
<script setup lang="ts">
import { isReasoningUIPart, isTextUIPart } from 'ai'
import { isPartStreaming } from '@nuxt/ui/utils/ai'
import shiki from '@comark/nuxt/plugins/shiki'

const plugins = [shiki()]
</script>

<template>
  <UChatMessages :messages="messages" :status="status">
    <template #content="{ message }">
      <template v-for="(part, index) in message.parts" :key="`${message.id}-${part.type}-${index}`">
        <UChatReasoning
          v-if="isReasoningUIPart(part)"
          :text="part.text"
          :streaming="isPartStreaming(part)"
        >
          <Markdown
            :value="part.text"
            :streaming="isPartStreaming(part)"
            :plugins="plugins"
            class="*:first:mt-0 *:last:mb-0"
          />
        </UChatReasoning>

        <template v-else-if="isTextUIPart(part)">
          <Markdown
            v-if="message.role === 'assistant'"
            :value="part.text"
            :streaming="isPartStreaming(part)"
            :plugins="plugins"
            class="*:first:mt-0 *:last:mb-0"
          />
          <p v-else-if="message.role === 'user'" class="whitespace-pre-wrap">
            {{ part.text }}
          </p>
        </template>
      </template>
    </template>
  </UChatMessages>
</template>
```
